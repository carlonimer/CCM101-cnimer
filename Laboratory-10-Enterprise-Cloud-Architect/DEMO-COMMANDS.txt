------------------------------------------------------------------------------
 PART C - PERSISTENCE DEMO (data survives container recreation / reboot)
------------------------------------------------------------------------------

# C1. Create a demo table and insert a row
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "CREATE TABLE IF NOT EXISTS demo(id serial PRIMARY KEY, note text);"'
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "INSERT INTO demo(note) VALUES (now()::text);"'
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "SELECT * FROM demo;"'

# C2. DESTROY and recreate every container (volume is kept)
docker compose down
docker compose up -d
docker compose ps

# C3. The row is still there -> persistence proven
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "SELECT * FROM demo;"'

# C4. (Optional, strongest proof) reboot the whole VM, then ssh back in and repeat C3
sudo reboot

------------------------------------------------------------------------------
 PART E - DISASTER RECOVERY (delete data, then restore from backup)
------------------------------------------------------------------------------

# E1. Simulate disaster: drop the demo table
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "DROP TABLE demo;"'

# E2. Pick the newest backup file
LATEST=$(ls -t backups/cloudapp_*.sql.gz | head -1); echo $LATEST

# E3. Restore it
gunzip -c $LATEST | docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB'

# E4. Data is back
docker compose exec -T db sh -c 'psql -U $POSTGRES_USER -d $POSTGRES_DB -c "SELECT * FROM demo;"'

NOTE: the backup in E2 must be taken AFTER step C1 (run D1 first) so it contains the demo table.
